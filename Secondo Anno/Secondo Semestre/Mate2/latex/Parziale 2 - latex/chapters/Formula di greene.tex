\chapter{Formula di Green}
\section{Cosa devo fare in questo esercizio}
Bisgona verificare la formula :
\begin{tcolorbox}[title={Formula di green}]
    {\large
    $$
    \oint_C \vec{F} \cdot \vec{t} \, dl = \iint_D (F_2)_x - (F_1)_y \, dxdy
    $$
    }
    \begin{itemize}
        \item $\vec{F}$ è campo vettoraile
        \item $\vec{t}$ è la derivata di $\vec{r}(t)$
        \item C è una curva
        \item D è il domio dell'integrale
        \item $(F_2)_x$,$ (F_1)_y$ sono le derivate delle componenti di $\vec{F}$
    \end{itemize}
\end{tcolorbox}

\section{Procedura}
\begin{enumerate}
    \item Verificare che il campo vettoriale sia conservatore 

    $$
    (F_1)_y = (F_2)_x \implies (F_1)_y - (F_2)_x = 0
    $$
    \begin{itemize}
        \item Se in caso il campo sia conservativo allora l'integrale doppio è pari a zero
    \end{itemize}

    \item Calcolare entrambi gli integrali e verificare che il loro risulato sia equivalente
\end{enumerate}
